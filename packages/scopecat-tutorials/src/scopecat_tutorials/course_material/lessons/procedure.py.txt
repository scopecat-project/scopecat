"""Editable synthetic procedure: acquire, ask a person, validate and finish.

The notebook uses task.pending_input(schema=PEAK_CHOICE) to read the current
question, including a later correction round. The choose-N keys below still
identify separate retained judgments; reading a question never starts a scan.
"""

from dataclasses import dataclass
from typing import Annotated, Literal, cast

from pydantic import BaseModel, ConfigDict

import scopecat as sc
from scopecat.api.procedures import InterpretationResult, LabProcedureContext
from scopecat.automation import AnalysisPublicationOutputRef, RunOutputRef
from scopecat.daemon.views import ParameterResolution
from scopecat.records.setup import SetupRevisionRef


class SurveyIntent(BaseModel):
    model_config = ConfigDict(extra="forbid", frozen=True)

    config: ParameterResolution
    setup: SetupRevisionRef


@sc.compute
def response(frequency: sc.Quantity) -> float:
    x = frequency.to("MHz").value
    return float(0.15 + 1 / (1 + ((x - 6500) / 1.2) ** 2))


@sc.experiment(id="teaching.procedure.synthetic-peak")
def sweep(
    _context: sc.ExperimentContext,
    frequency: Annotated[
        sc.Input[sc.Quantity], sc.ControlSpec(unit="MHz", scannable=True)
    ],
) -> dict[str, object]:
    return {"frequency": frequency, "response": response(frequency)}


@dataclass(frozen=True)
class CurvePoint:
    frequency: Annotated[float, sc.AnalysisField(unit="MHz", role="coordinate")]
    response: Annotated[float, sc.AnalysisField()]


@sc.analysis_step(id="teaching.procedure.peak-curve")
def curve(context: sc.AnalysisContext) -> sc.Analysis:
    data = context.measurements()
    x = cast(
        "list[float]", data["frequency"].dense.to("MHz").values.reshape(-1).tolist()
    )
    y = cast("list[float]", data["response"].dense.values.reshape(-1).tolist())
    points = [CurvePoint(a, b) for a, b in zip(x, y, strict=True)]
    return (
        context.result("Synthetic response — no device connected")
        .dataset("curve", points)
        .figure(dataset="curve", id="trace", kind="line", x="frequency", y="response")
    )


@dataclass(frozen=True)
class PeakChoice:
    outcome: Literal["Peak selected", "No clear peak", "Unsure"]
    frequency_mhz: float | None


MIN_FREQUENCY_MHZ = 6490
MAX_FREQUENCY_MHZ = 6510

PEAK_CHOICE = sc.AnalysisFactSchema("teaching.procedure.peak-choice.v1", PeakChoice)


def peak_choice_problem(value: PeakChoice) -> str:
    """Check the answer's range and combination, not whether it is the right peak."""
    if value.outcome == "Peak selected":
        if value.frequency_mhz is None:
            return "Enter a peak frequency when choosing Peak selected."
        if not MIN_FREQUENCY_MHZ <= value.frequency_mhz <= MAX_FREQUENCY_MHZ:
            return (
                f"Peak frequency must be between {MIN_FREQUENCY_MHZ} "
                f"and {MAX_FREQUENCY_MHZ} MHz."
            )
    elif value.frequency_mhz is not None:
        return "Leave the frequency empty for No clear peak or Unsure."
    return ""


def choose_peak(
    context: LabProcedureContext,
    scan: RunOutputRef,
    plotted: AnalysisPublicationOutputRef,
) -> InterpretationResult[PeakChoice]:
    """Ask about these retained results until the answer satisfies our rules.

    Called once by survey. Each choose-N is a new judgment round, not a retry
    attempt of an existing step. Replay reads the same answers and never scans.
    Return the whole judgment so later author steps can retain choice.ref.
    """
    round_index = 1
    problem = ""
    while True:
        choice = context.interpret(
            f"choose-{round_index}",
            title="Choose a peak frequency",
            instructions=(
                "Find the highest point of the upward peak (not a dip). "
                "Hover over the curve to read its horizontal coordinate, "
                "then type the frequency; clicking does not fill it. "
                "Enter a frequency in MHz between "
                f"{MIN_FREQUENCY_MHZ} and {MAX_FREQUENCY_MHZ}, "
                "or choose No clear peak / Unsure and leave it empty. "
                "Record your answer and explicitly resume the procedure to finish. "
                "Completion records your judgment, "
                "not its correctness or a published calibration. " + problem
            ),
            schema=PEAK_CHOICE,
            inputs=(scan, plotted),
            response_template=PeakChoice("Unsure", None),
            metadata={
                "field_labels": {
                    "outcome": "Your judgment",
                    "frequency_mhz": "Peak frequency (MHz)",
                },
                "field_constraints": {
                    "frequency_mhz": {
                        "minimum": MIN_FREQUENCY_MHZ,
                        "maximum": MAX_FREQUENCY_MHZ,
                        "required_when": {"field": "outcome", "value": "Peak selected"},
                        "empty_otherwise": True,
                    }
                },
            },
        )
        problem = peak_choice_problem(choice.value)
        if not problem:
            return choice
        round_index += 1


@sc.procedure(id="teaching.procedure.manual-peaks", version="1", intent=SurveyIntent)
def survey(context: LabProcedureContext, intent: SurveyIntent) -> None:
    scan = context.run(
        "scan",
        sweep(frequency=sc.Quantity(6500, "MHz")).sweep(
            frequency=[sc.Quantity(6490 + index * 0.25, "MHz") for index in range(81)]
        ),
        config=intent.config,
        setup=intent.setup,
        name="Procedure lesson: synthetic frequency scan",
    )
    plotted = context.analyze_run("curve", scan, curve())
    choose_peak(context, scan, plotted)
