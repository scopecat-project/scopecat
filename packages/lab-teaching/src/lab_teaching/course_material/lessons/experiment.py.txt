"""Synthetic course experiment; uses the selected parameter workspace."""

from typing import Annotated

import scopecat as sc
from my_experiment.parameters import Drive
from my_experiment.response import response


@sc.experiment(id="teaching.rabi")
def teaching_rabi(
    experiment: sc.ExperimentContext,
    shots: int = 64,
    seed: int = 200,
    no_response: bool = False,
    *,
    amplitude: Annotated[
        sc.Input[sc.Quantity],
        sc.ControlSpec(minimum=0, maximum=0.9, title="Drive amplitude", scannable=True),
    ] = sc.Quantity(0.1, "arb"),
) -> dict[str, object]:
    iq = response(
        frequency=sc.parameter_ref(Drive.frequency, "q0"),
        amplitude=amplitude,
        shots=shots,
        seed=seed,
        no_response=no_response,
    )
    return {"amplitude": experiment.coordinate(amplitude), "iq": iq}
